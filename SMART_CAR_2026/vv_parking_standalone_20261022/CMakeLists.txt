cmake_minimum_required(VERSION 3.16)
project(vv_parking_standalone LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(VV_HOST_TESTS_ONLY "Build hardware-independent controller tests only" OFF)
include(CTest)
if(BUILD_TESTING)
    add_executable(standalone_flow_test tests/standalone_flow_test.cpp)
    target_include_directories(standalone_flow_test PRIVATE vv/src)
    if(NOT MSVC)
        target_compile_options(standalone_flow_test PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
    endif()
    add_test(NAME standalone_flow COMMAND standalone_flow_test)
endif()
if(NOT VV_HOST_TESTS_ONLY)
    find_package(OpenCV REQUIRED)
    find_package(Threads REQUIRED)
    file(GLOB_RECURSE VV_SOURCES CONFIGURE_DEPENDS "vv/src/*.cpp" "vv/src/*.c")
    list(REMOVE_ITEM VV_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/vv/src/main.cpp")
    add_executable(vv_parking_test standalone/main.cpp ${VV_SOURCES})
    target_include_directories(vv_parking_test PRIVATE vv/src vv/src/Experiment/Baselines ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(vv_parking_test PRIVATE ${OpenCV_LIBS} Threads::Threads)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
        target_link_libraries(vv_parking_test PRIVATE stdc++fs)
    endif()
    set_target_properties(vv_parking_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
