cmake_minimum_required(VERSION 3.16)
project(SMART_CAR_2026 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_VEHICLE "Build Linux camera and Loongson hardware executable" ${CMAKE_HOST_UNIX})
add_library(car_core src/config.cpp src/vision.cpp src/mission.cpp src/imu.cpp)
target_include_directories(car_core PUBLIC src)
if(MSVC)
  target_compile_options(car_core PRIVATE /W4)
else()
  target_compile_options(car_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(vision_stream src/replay_stream.cpp)
target_link_libraries(vision_stream PRIVATE car_core)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE car_core)
add_library(car_hardware src/hardware.cpp)
target_link_libraries(car_hardware PUBLIC car_core)
if(MSVC)
  target_compile_options(car_hardware PRIVATE /W4)
else()
  target_compile_options(car_hardware PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(hardware_tests tests/hardware_tests.cpp)
target_link_libraries(hardware_tests PRIVATE car_hardware)
add_executable(capture_data_tests tests/capture_data_tests.cpp)
target_include_directories(capture_data_tests PRIVATE src)
enable_testing()
add_test(NAME core_tests COMMAND core_tests ${CMAKE_CURRENT_SOURCE_DIR}/config/competition.ini)
add_test(NAME hardware_tests COMMAND hardware_tests ${CMAKE_CURRENT_SOURCE_DIR}/config/hardware.ini)
add_test(NAME capture_data_tests COMMAND capture_data_tests)
if(UNIX)
  add_library(car_linux_devices src/hardware_linux.cpp)
  target_link_libraries(car_linux_devices PUBLIC car_hardware)
  target_compile_options(car_linux_devices PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(hardware_servo_test
    tools/hardware_servo_test.cpp)
  target_link_libraries(hardware_servo_test PRIVATE car_linux_devices)
  target_compile_options(hardware_servo_test PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(hardware_pwm_probe tools/hardware_pwm_probe.cpp)
  target_link_libraries(hardware_pwm_probe PRIVATE car_linux_devices)
  target_compile_options(hardware_pwm_probe PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_VEHICLE)
  if(NOT UNIX)
    message(FATAL_ERROR "Vehicle drivers require Linux. Use -DBUILD_VEHICLE=OFF on Windows.")
  endif()
  find_package(OpenCV REQUIRED COMPONENTS core imgproc videoio highgui)
  find_package(Threads REQUIRED)
  add_executable(manual_capture tools/manual_capture.cpp)
  target_include_directories(manual_capture PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(manual_capture PRIVATE car_linux_devices ${OpenCV_LIBS} Threads::Threads)
  target_compile_options(manual_capture PRIVATE -Wall -Wextra -Wpedantic)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
    target_link_libraries(manual_capture PRIVATE stdc++fs)
  endif()
  # Only the hardware-independent PID remains from legacy. No old sysfs or /dev/mem drivers.
  add_executable(SMART_CAR_2026 src/linux_main.cpp
    legacy/Contral/PID/PID.c)
  target_include_directories(SMART_CAR_2026 PRIVATE legacy ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(SMART_CAR_2026 PRIVATE car_linux_devices ${OpenCV_LIBS} Threads::Threads m)
endif()
