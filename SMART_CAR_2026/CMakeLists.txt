cmake_minimum_required(VERSION 3.16)
project(SMART_CAR_2026 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_VEHICLE "Build Linux camera and Loongson hardware executable" ${CMAKE_HOST_UNIX})
add_library(car_core src/config.cpp src/vision.cpp src/mission.cpp src/imu.cpp src/visual_observer.cpp)
target_include_directories(car_core PUBLIC src)
if(MSVC)
  target_compile_options(car_core PRIVATE /W4)
else()
  target_compile_options(car_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(vision_stream src/replay_stream.cpp)
target_link_libraries(vision_stream PRIVATE car_core)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE car_core)
add_executable(visual_observer_tests tests/visual_observer_tests.cpp)
target_link_libraries(visual_observer_tests PRIVATE car_core)
# Standalone metric parking model/controller. No camera, IMU or actuator library.
add_library(car_parking src/reverse_parking.cpp)
target_include_directories(car_parking PUBLIC src)
if(MSVC)
  target_compile_options(car_parking PRIVATE /W4)
else()
  target_compile_options(car_parking PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(reverse_parking_tests tests/reverse_parking_tests.cpp)
target_link_libraries(reverse_parking_tests PRIVATE car_parking)
add_executable(parking_geometry_check tools/parking_geometry_check.cpp)
target_link_libraries(parking_geometry_check PRIVATE car_parking)
add_library(car_hardware src/hardware.cpp)
target_link_libraries(car_hardware PUBLIC car_core)
if(MSVC)
  target_compile_options(car_hardware PRIVATE /W4)
else()
  target_compile_options(car_hardware PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(hardware_tests tests/hardware_tests.cpp)
target_link_libraries(hardware_tests PRIVATE car_hardware)
add_executable(capture_data_tests tests/capture_data_tests.cpp)
target_include_directories(capture_data_tests PRIVATE src)
add_executable(manual_steering_tests tests/manual_steering_tests.cpp)
target_link_libraries(manual_steering_tests PRIVATE car_hardware)
add_executable(rehearsal_servo_tests tests/rehearsal_servo_tests.cpp)
find_package(Threads REQUIRED)
target_link_libraries(rehearsal_servo_tests PRIVATE car_hardware Threads::Threads)
add_executable(parking_rehearsal_tests tests/parking_rehearsal_tests.cpp)
target_include_directories(parking_rehearsal_tests PRIVATE src)
add_executable(rehearsal_speed_tests tests/rehearsal_speed_tests.cpp)
target_link_libraries(rehearsal_speed_tests PRIVATE car_hardware)
enable_testing()
add_test(NAME rehearsal_servo_tests COMMAND rehearsal_servo_tests)
add_test(NAME parking_rehearsal_tests COMMAND parking_rehearsal_tests)
add_test(NAME rehearsal_speed_tests COMMAND rehearsal_speed_tests)
add_test(NAME core_tests COMMAND core_tests ${CMAKE_CURRENT_SOURCE_DIR}/config/competition.ini)
add_test(NAME visual_observer_tests COMMAND visual_observer_tests)
add_test(NAME reverse_parking_tests COMMAND reverse_parking_tests)
add_test(NAME hardware_tests COMMAND hardware_tests ${CMAKE_CURRENT_SOURCE_DIR}/config/hardware.ini)
add_test(NAME capture_data_tests COMMAND capture_data_tests)
add_test(NAME manual_steering_tests COMMAND manual_steering_tests)
add_executable(gray_dt_ranges_tests tests/gray_dt_ranges_tests.c)
set_target_properties(gray_dt_ranges_tests PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
if(MSVC)
  target_compile_options(gray_dt_ranges_tests PRIVATE /W4)
else()
  target_compile_options(gray_dt_ranges_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME gray_dt_ranges_tests COMMAND gray_dt_ranges_tests)
if(UNIX)
  add_executable(gray_gpio_inventory tools/gray_gpio_inventory.c)
  set_target_properties(gray_gpio_inventory PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
  target_compile_options(gray_gpio_inventory PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(gray_gpio_levels tools/gray_gpio_levels.c)
  set_target_properties(gray_gpio_levels PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
  target_compile_options(gray_gpio_levels PRIVATE -Wall -Wextra -Wpedantic)
  add_library(car_linux_devices src/hardware_linux.cpp)
  target_link_libraries(car_linux_devices PUBLIC car_hardware)
  target_compile_options(car_linux_devices PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(hardware_servo_test
    tools/hardware_servo_test.cpp)
  target_link_libraries(hardware_servo_test PRIVATE car_linux_devices)
  target_compile_options(hardware_servo_test PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(hardware_pwm_probe tools/hardware_pwm_probe.cpp)
  target_link_libraries(hardware_pwm_probe PRIVATE car_linux_devices)
  target_compile_options(hardware_pwm_probe PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(hardware_bench_test tools/hardware_bench_test.cpp)
  target_link_libraries(hardware_bench_test PRIVATE car_linux_devices Threads::Threads)
  target_compile_options(hardware_bench_test PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_VEHICLE)
  if(NOT UNIX)
    message(FATAL_ERROR "Vehicle drivers require Linux. Use -DBUILD_VEHICLE=OFF on Windows.")
  endif()
  find_package(OpenCV REQUIRED COMPONENTS core imgproc videoio highgui)
  add_executable(parking_rehearsal tools/parking_rehearsal.cpp)
  target_include_directories(parking_rehearsal PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(parking_rehearsal PRIVATE car_linux_devices opencv_core opencv_videoio Threads::Threads)
  target_compile_options(parking_rehearsal PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(manual_capture tools/manual_capture.cpp)
  target_include_directories(manual_capture PRIVATE ${OpenCV_INCLUDE_DIRS})
  # Image processing and AVI output only; no GUI/display calls in this tool.
  target_link_libraries(manual_capture PRIVATE car_linux_devices opencv_core opencv_imgproc opencv_videoio Threads::Threads)
  target_compile_options(manual_capture PRIVATE -Wall -Wextra -Wpedantic)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
    target_link_libraries(manual_capture PRIVATE stdc++fs)
    target_link_libraries(parking_rehearsal PRIVATE stdc++fs)
  endif()
  # Only the hardware-independent PID remains from legacy. No old sysfs or /dev/mem drivers.
  add_executable(SMART_CAR_2026 src/linux_main.cpp
    legacy/Contral/PID/PID.c)
  target_include_directories(SMART_CAR_2026 PRIVATE legacy ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(SMART_CAR_2026 PRIVATE car_linux_devices ${OpenCV_LIBS} Threads::Threads m)
endif()
